% data/skills.tex
% Defines all role-specific skill set commands used in
% sections/skills.tex.
%
% HOW IT WORKS:
% Each \newcommand defines one complete skill set.
% Inside each command, \SkillCategory{label}{skills} renders
% one row: bold category label followed by comma-separated skills.
%
% To update skills for a role — edit the relevant command below.
% To add a new role:
%   1. Add \newcommand{\XxxSkills}{...} here
%   2. Add \IfXxx{\XxxSkills} in sections/skills.tex
%   3. Add \IfXxx{} in settings.tex
% ============================================================
