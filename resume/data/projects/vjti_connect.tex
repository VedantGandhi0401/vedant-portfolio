% Consulting because a social networking platform is not
% relevant signal for those role types.
%
% To update a bullet — edit the relevant command below.
% To add a new role variant:
%   1. Add \newcommand{\VJTIConnectBulletXxx}{...} here
%   2. Add \IfXxx{\VJTIConnectBulletXxx} in sections/projects.tex
% ============================================================

% ============================================================
% SDE — Software Development Engineer
% Signals: full-stack ownership, secure auth, REST API design,
% and scalable modular architecture on a real platform.
% ============================================================
\newcommand{\VJTIConnectBulletSDE}{
  \item Built a full-stack social networking platform
        exclusively for VJTI's 4,500+ student campus using
        the MERN stack, with JWT authentication and
        role-based access control.
  \item Designed REST APIs supporting posts, real-time
        messaging via Socket.IO, comments, notifications
        and user profiles.
  \item Structured the backend for horizontal scalability,
        with independently deployable service modules
        enabling future feature additions without
        cross-cutting changes.
}


% ============================================================
% Backend
% Signals: Express.js API design, MongoDB schema optimization,
% JWT auth implementation, and scalable service structure.
% ============================================================
\newcommand{\VJTIConnectBulletBackend}{
  \item Designed REST APIs using Express.js covering
        authentication, real-time messaging (Socket.IO),
        social feeds and notifications for VJTI's
        4,500+ student campus.
  \item Optimized MongoDB schemas and query patterns for
        read-heavy social feed and notification workloads.
  \item Enforced JWT authentication and role-based access
        control across all protected API routes.
}


% ============================================================
% Frontend
% Signals: React + TypeScript UI, reusable components,
% async data fetching, and state management.
% ============================================================
\newcommand{\VJTIConnectBulletFrontend}{
  \item Built a React + TypeScript UI for VJTI's campus
        social platform, covering profiles, feeds,
        real-time messaging and notifications.
  \item Integrated frontend with backend REST APIs using
        async data fetching, optimistic UI updates and
        structured error handling.
  \item Improved perceived performance through component-
        level code splitting and efficient state management.
}


% ============================================================
% Full Stack
% Signals: end-to-end MERN ownership, auth, REST APIs,
% and scalable architecture across both layers.
% ============================================================
\newcommand{\VJTIConnectBulletFullStack}{
  \item Built a full-stack social networking platform for
        VJTI students using the MERN stack end-to-end.
  \item Developed React frontend interfaces and Express.js
        backend APIs for complete feature coverage.
  \item Implemented JWT authentication, MongoDB integration,
        and RESTful API architecture.
  \item Designed scalable project architecture supporting
        future feature additions and user growth.
}

% Security
\newcommand{\VJTIConnectBulletSecurity}{
  \item Implemented JWT authentication and role-based access
        control across all protected API routes, with input
        validation and rate limiting on public endpoints.
  \item Applied OWASP principles to the authentication layer:

        protection against brute-force login attempts.
  \item Structured error responses to avoid leaking
        implementation details to unauthenticated clients.
