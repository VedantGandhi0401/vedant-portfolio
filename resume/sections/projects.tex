% sections/projects.tex
% Renders project entries, capped at \MaxProjects (config.tex).
%
% HOW IT WORKS:
% 1. \input{} at the top loads project bullet definitions.
% 2. \ProjectSlot{prefix}{block} renders #2 only if a bullet
%    variant named \<prefix><Role> exists AND fewer than
%    \MaxProjects entries have rendered so far for this role.
% 3. Blocks are ordered by PRIORITY -- the project listed
%    first gets first claim on a slot. Reorder the blocks
%    below to change priority; \MaxProjects in config.tex
%    controls how many slots exist.
%
% CURRENT PRIORITY ORDER:
%   1. CBASNet
%   2. Investment Research Report Generator
%   3. Crop Change Detection
%   4. VJTI Connect
%   5. Community Waste Management Platform
%   6. LogicBlocks
%   7. Policy Tracker (In Progress)
%
% To add a new project:
%   1. Create data/projects/yourproject.tex, defining bullet
%      variants named \YourProjectBullet<Role> (or whatever
%      prefix you choose) for each role it's relevant to.
%   2. \input{} it below.
%   3. Add a \ProjectSlot{prefix}{ \ProjectEntry{...} ... }
%      block wherever it should rank in priority order.
% ============================================================
